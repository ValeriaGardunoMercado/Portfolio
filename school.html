Global Integer num_base
Global String lado_fus$
Global Integer torre_fus
Global String ranura$


' Programa principal
Function main
    Reset
    Motor On
    Power Low
    Home
    
    String caja_lado$
    Integer i
    String repetir$
    
    Do
        ' Pregunta lado de Caja
        Print "Cual caja quieres? (D / I)"
        Input caja_lado$
        
        If caja_lado$ = "D" Then
            Call toma_caja_derecha
        ElseIf caja_lado$ = "I" Then
            Call toma_caja_izquierda
        EndIf
        
        ' Pregunta numero de Base
        Print "En cual base? (1, 2, 3, 4, 5, 6)"
        Input num_base
        Call colocar_caja
        
        'Preguntas de Fusibles (2 veces)
        For i = 1 To 2
            Print "--- Fusible ", i, " ---"
            Print "De que lado tomo el fusible? (D o I)"
            Input lado_fus$
            
            Print "De que torre tomo el fusible? (1, 2, 3)"
            Input torre_fus
            
            Call toma_fusible
            
            If i = 1 Then
                ranura$ = "I"
            Else
                ranura$ = "D"
            EndIf
            
            Call colocar_fusible
        Next i
        
        Home
        Print "Ciclo de ensamble completado."
        
        ' --- 4. Pregunta para repetir ---
        Print "Deseas realizar el proceso de nuevo? (si / no)"
        Input repetir$
        
    Loop While repetir$ = "si" Or repetir$ = "s" Or repetir$ = "SI"
    
    Print "Programa finalizado correctamente."
Fend

Function toma_caja_derecha
    On 2
    Pass Caja_Der_4
    Pass Caja_Der_3
    Pass Caja_Der_2
    Go Caja_Der_1
    Off 2
    Wait 0.2
    Pass Caja_Der_2

Fend
